// Allergies page

import 'package:flutter/material.dart';

import '../models.dart';
import 'goal_page.dart';

const _green = Color(0xFF138A3E);

class AllergiesPage extends StatefulWidget {
  final OnboardingData data;
  const AllergiesPage({super.key, required this.data});
  @override
  State<AllergiesPage> createState() => _AllergiesPageState();
}

class _AllergiesPageState extends State<AllergiesPage> {
  String query = '';
  OnboardingData get d => widget.data;

  bool get hasHighAlert =>
      allergens.any((a) => a.highAlert && d.allergies.contains(a.name));

  void toggle(String name) => setState(() {
    d.allergies.contains(name)
        ? d.allergies.remove(name)
        : d.allergies.add(name);
    d.noAllergies = false;
  });

  void toggleNone() => setState(() {
    d.noAllergies = !d.noAllergies;
    if (d.noAllergies) d.allergies.clear();
  });

  Widget _card({
    required Widget child,
    bool selected = false,
    VoidCallback? onTap,
  }) => InkWell(
    onTap: onTap,
    borderRadius: BorderRadius.circular(16),
    child: Container(
      padding: const EdgeInsets.all(12),
      decoration: BoxDecoration(
        color: selected ? const Color(0xFFEAF8EE) : Colors.white,
        borderRadius: BorderRadius.circular(16),
        border: Border.all(
          color: selected ? _green : Colors.grey.shade300,
          width: selected ? 1.5 : 1,
        ),
      ),
      child: child,
    ),
  );

  @override
  Widget build(BuildContext context) {
    final list = allergens
        .where((a) => a.name.toLowerCase().contains(query.toLowerCase()))
        .toList();

    return Scaffold(
      backgroundColor: const Color(0xFFF6F6FD),
      body: SafeArea(
        child: Column(
          children: [
            // Header
            Padding(
              padding: const EdgeInsets.fromLTRB(8, 8, 16, 0),
              child: Row(
                children: [
                  const IconButton(
                    icon: Icon(Icons.arrow_back),
                    onPressed: null,
                  ),
                  Expanded(
                    child: Row(
                      children: List.generate(
                        4,
                        (i) => Expanded(
                          child: Container(
                            height: 4,
                            margin: const EdgeInsets.symmetric(horizontal: 2),
                            decoration: BoxDecoration(
                              color: i < 1 ? _green : Colors.grey.shade300,
                              borderRadius: BorderRadius.circular(2),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
                  const SizedBox(width: 8),
                  const Text('1 / 4', style: TextStyle(fontSize: 12)),
                ],
              ),
            ),
            // Body
            Expanded(
              child: SingleChildScrollView(
                padding: const EdgeInsets.all(16),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text(
                      'Any food allergies or intolerances?',
                      style: TextStyle(
                        fontSize: 24,
                        fontWeight: FontWeight.w800,
                      ),
                    ),
                    const SizedBox(height: 4),
                    Text(
                      'NutriLens AI will instantly flag unsafe ingredients during camera scanning.',
                      style: TextStyle(color: Colors.grey.shade600),
                    ),
                    const SizedBox(height: 16),
                    TextField(
                      onChanged: (v) => setState(() => query = v),
                      decoration: InputDecoration(
                        hintText: 'Search allergens (e.g. peanuts, gluten...)',
                        prefixIcon: const Icon(Icons.search),
                        filled: true,
                        fillColor: Colors.white,
                        border: OutlineInputBorder(
                          borderRadius: BorderRadius.circular(14),
                          borderSide: BorderSide.none,
                        ),
                      ),
                    ),
                    const SizedBox(height: 16),
                    Row(
                      mainAxisAlignment: MainAxisAlignment.spaceBetween,
                      children: [
                        const Text(
                          'Common Allergens',
                          style: TextStyle(fontWeight: FontWeight.w600),
                        ),
                        Text(
                          '${d.allergies.length} Selected',
                          style: const TextStyle(
                            color: _green,
                            fontSize: 12,
                            fontWeight: FontWeight.w600,
                          ),
                        ),
                      ],
                    ),
                    const SizedBox(height: 8),
                    GridView.count(
                      crossAxisCount: 3,
                      shrinkWrap: true,
                      physics: const NeverScrollableScrollPhysics(),
                      mainAxisSpacing: 8,
                      crossAxisSpacing: 8,
                      childAspectRatio: 0.95,
                      children: [
                        for (final a in list)
                          _card(
                            selected: d.allergies.contains(a.name),
                            onTap: () => toggle(a.name),
                            child: Column(
                              mainAxisAlignment: MainAxisAlignment.center,
                              children: [
                                Icon(
                                  a.icon,
                                  color: d.allergies.contains(a.name)
                                      ? _green
                                      : Colors.grey,
                                ),
                                const SizedBox(height: 6),
                                Text(
                                  a.name,
                                  style: const TextStyle(
                                    fontWeight: FontWeight.w700,
                                    fontSize: 13,
                                  ),
                                ),
                                Text(
                                  a.note,
                                  style: TextStyle(
                                    fontSize: 10,
                                    color: a.highAlert
                                        ? _green
                                        : Colors.grey.shade600,
                                  ),
                                ),
                              ],
                            ),
                          ),
                      ],
                    ),
                    const SizedBox(height: 12),
                    _card(
                      onTap: toggleNone,
                      child: Row(
                        children: [
                          const Icon(Icons.check_circle_outline),
                          const SizedBox(width: 10),
                          const Expanded(
                            child: Text('No allergies or dietary restrictions'),
                          ),
                          Checkbox(
                            value: d.noAllergies,
                            onChanged: (_) => toggleNone(),
                          ),
                        ],
                      ),
                    ),
                    if (hasHighAlert) ...[
                      const SizedBox(height: 12),
                      _card(
                        child: const Row(
                          children: [
                            Icon(
                              Icons.warning_amber_rounded,
                              color: Colors.deepOrange,
                            ),
                            SizedBox(width: 10),
                            Expanded(
                              child: Text(
                                'Alert Mode: Strict Warning — scanning will flag any match with red alerts.',
                                style: TextStyle(fontSize: 12),
                              ),
                            ),
                            Text(
                              'ACTIVE',
                              style: TextStyle(
                                color: _green,
                                fontSize: 11,
                                fontWeight: FontWeight.w700,
                              ),
                            ),
                          ],
                        ),
                      ),
                    ],
                  ],
                ),
              ),
            ),
            // Button
            Padding(
              padding: const EdgeInsets.fromLTRB(16, 8, 16, 12),
              child: SizedBox(
                width: double.infinity,
                height: 52,
                child: FilledButton(
                  style: FilledButton.styleFrom(backgroundColor: _green),
                  onPressed: () => Navigator.push(
                    context,
                    MaterialPageRoute(builder: (_) => GoalPage(data: d)),
                  ),
                  child: const Text(
                    'Complete Setup',
                    style: TextStyle(fontSize: 16),
                  ),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}


// body profile page
import 'package:flutter/material.dart';

import '../models.dart';
import 'connect_doctor_page.dart';

const _green = Color(0xFF138A3E);

class BodyProfilePage extends StatefulWidget {
  final OnboardingData data;
  const BodyProfilePage({super.key, required this.data});
  @override
  State<BodyProfilePage> createState() => _BodyProfilePageState();
}

class _BodyProfilePageState extends State<BodyProfilePage> {
  OnboardingData get d => widget.data;
  bool useCm = true;
  bool useKg = true;

  static const lbsPerKg = 2.20462;

  String get heightText {
    if (useCm) return '${d.heightCm.round()} cm';
    final totalIn = d.heightCm / 2.54;
    return "${totalIn ~/ 12}' ${(totalIn % 12).round()}\"";
  }

  String get weightText => useKg
      ? '${d.weightKg.toStringAsFixed(1)} kg'
      : '${(d.weightKg * lbsPerKg).toStringAsFixed(1)} lbs';

  void changeWeight(double delta) => setState(() {
    final kgDelta = useKg ? delta : delta / lbsPerKg;
    d.weightKg = (d.weightKg + kgDelta).clamp(20.0, 300.0);
  });

  Widget _card({
    required Widget child,
    bool selected = false,
    VoidCallback? onTap,
  }) => InkWell(
    onTap: onTap,
    borderRadius: BorderRadius.circular(16),
    child: Container(
      padding: const EdgeInsets.all(12),
      decoration: BoxDecoration(
        color: selected ? const Color(0xFFEAF8EE) : Colors.white,
        borderRadius: BorderRadius.circular(16),
        border: Border.all(
          color: selected ? _green : Colors.grey.shade300,
          width: selected ? 1.5 : 1,
        ),
      ),
      child: child,
    ),
  );

  Widget _toggle(
    List<String> options,
    bool first,
    ValueChanged<bool> onChanged,
  ) => ToggleButtons(
    isSelected: [first, !first],
    onPressed: (i) => onChanged(i == 0),
    constraints: const BoxConstraints(minHeight: 30, minWidth: 48),
    borderRadius: BorderRadius.circular(8),
    selectedColor: Colors.white,
    fillColor: _green,
    children: options
        .map((o) => Text(o, style: const TextStyle(fontSize: 12)))
        .toList(),
  );

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFF6F6FD),
      body: SafeArea(
        child: Column(
          children: [
            Padding(
              padding: const EdgeInsets.fromLTRB(8, 8, 16, 0),
              child: Row(
                children: [
                  IconButton(
                    icon: const Icon(Icons.arrow_back),
                    onPressed: () => Navigator.pop(context),
                  ),
                  Expanded(
                    child: Row(
                      children: List.generate(
                        4,
                        (i) => Expanded(
                          child: Container(
                            height: 4,
                            margin: const EdgeInsets.symmetric(horizontal: 2),
                            decoration: BoxDecoration(
                              color: i < 3 ? _green : Colors.grey.shade300,
                              borderRadius: BorderRadius.circular(2),
                            ),
                          ),
                        ),
                      ),
                    ),
                  ),
                  const SizedBox(width: 8),
                  const Text('3 / 4', style: TextStyle(fontSize: 12)),
                ],
              ),
            ),
            Expanded(
              child: SingleChildScrollView(
                padding: const EdgeInsets.all(16),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text(
                      "What's your body profile?",
                      style: TextStyle(
                        fontSize: 24,
                        fontWeight: FontWeight.w800,
                      ),
                    ),
                    const SizedBox(height: 4),
                    Text(
                      'This helps NutriLens calculate your baseline calorie and metabolic needs.',
                      style: TextStyle(color: Colors.grey.shade600),
                    ),
                    const SizedBox(height: 16),

                    // Sex
                    const Text(
                      'Biological Sex',
                      style: TextStyle(fontWeight: FontWeight.w600),
                    ),
                    const SizedBox(height: 8),
                    Row(
                      children: [
                        for (final s in Sex.values)
                          Expanded(
                            child: Padding(
                              padding: const EdgeInsets.only(right: 8),
                              child: _card(
                                selected: d.sex == s,
                                onTap: () => setState(() => d.sex = s),
                                child: Column(
                                  children: [
                                    Icon(
                                      switch (s) {
                                        Sex.male => Icons.male,
                                        Sex.female => Icons.female,
                                        Sex.neutral => Icons.transgender,
                                      },
                                      color: d.sex == s ? _green : Colors.grey,
                                    ),
                                    Text(
                                      s.name[0].toUpperCase() +
                                          s.name.substring(1),
                                      style: const TextStyle(
                                        fontSize: 12,
                                        fontWeight: FontWeight.w600,
                                      ),
                                    ),
                                  ],
                                ),
                              ),
                            ),
                          ),
                      ],
                    ),
                    const SizedBox(height: 12),

                    // Height
                    _card(
                      child: Column(
                        children: [
                          Row(
                            children: [
                              const Expanded(
                                child: Text(
                                  'Height',
                                  style: TextStyle(fontWeight: FontWeight.w700),
                                ),
                              ),
                              _toggle(
                                ['cm', 'ft/in'],
                                useCm,
                                (v) => setState(() => useCm = v),
                              ),
                            ],
                          ),
                          Text(
                            heightText,
                            style: const TextStyle(
                              fontSize: 32,
                              fontWeight: FontWeight.w800,
                            ),
                          ),
                          Slider(
                            value: d.heightCm,
                            min: 120,
                            max: 220,
                            activeColor: _green,
                            onChanged: (v) => setState(() => d.heightCm = v),
                          ),
                        ],
                      ),
                    ),
                    const SizedBox(height: 12),

                    // Weight
                    _card(
                      child: Column(
                        children: [
                          Row(
                            children: [
                              const Expanded(
                                child: Text(
                                  'Current Weight',
                                  style: TextStyle(fontWeight: FontWeight.w700),
                                ),
                              ),
                              _toggle(
                                ['kg', 'lbs'],
                                useKg,
                                (v) => setState(() => useKg = v),
                              ),
                            ],
                          ),
                          const SizedBox(height: 8),
                          Row(
                            mainAxisAlignment: MainAxisAlignment.spaceEvenly,
                            children: [
                              IconButton.outlined(
                                onPressed: () => changeWeight(-0.5),
                                icon: const Icon(Icons.remove),
                              ),
                              Text(
                                weightText,
                                style: const TextStyle(
                                  fontSize: 28,
                                  fontWeight: FontWeight.w800,
                                ),
                              ),
                              IconButton.filled(
                                style: IconButton.styleFrom(
                                  backgroundColor: _green,
                                ),
                                onPressed: () => changeWeight(0.5),
                                icon: const Icon(Icons.add),
                              ),
                            ],
                          ),
                          const SizedBox(height: 8),
                          Wrap(
                            spacing: 8,
                            children: [
                              for (final delta in const [-2.0, -0.5, 0.5, 2.0])
                                ActionChip(
                                  label: Text(delta > 0 ? '+$delta' : '$delta'),
                                  onPressed: () => changeWeight(delta),
                                ),
                            ],
                          ),
                        ],
                      ),
                    ),
                    const SizedBox(height: 12),

                    // Age
                    _card(
                      child: Row(
                        children: [
                          const Expanded(
                            child: Text(
                              'Age',
                              style: TextStyle(fontWeight: FontWeight.w700),
                            ),
                          ),
                          IconButton(
                            onPressed: () => setState(
                              () => d.age = (d.age - 1).clamp(10, 100),
                            ),
                            icon: const Icon(Icons.remove_circle_outline),
                          ),
                          Text(
                            '${d.age} yrs',
                            style: const TextStyle(
                              fontSize: 18,
                              fontWeight: FontWeight.w700,
                            ),
                          ),
                          IconButton(
                            onPressed: () => setState(
                              () => d.age = (d.age + 1).clamp(10, 100),
                            ),
                            icon: const Icon(Icons.add_circle_outline),
                          ),
                        ],
                      ),
                    ),
                  ],
                ),
              ),
            ),
            Padding(
              padding: const EdgeInsets.fromLTRB(16, 8, 16, 12),
              child: SizedBox(
                width: double.infinity,
                height: 52,
                child: FilledButton(
                  style: FilledButton.styleFrom(backgroundColor: _green),
                  onPressed: () => Navigator.push(
                    context,
                    MaterialPageRoute(builder: (_) => DoctorPage(data: d)),
                  ),
                  child: const Text('Continue', style: TextStyle(fontSize: 16)),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class DoctorPage extends StatefulWidget {
  final OnboardingData data;
  const DoctorPage({super.key, required this.data});
  @override
  State<DoctorPage> createState() => _DoctorPageState();
}

class _DoctorPageState extends State<DoctorPage> {
  final controller = TextEditingController();
  Doctor? found;
  bool loading = false;
  bool notFound = false;

  OnboardingData get d => widget.data;

  Future<void> apply() async {
    setState(() {
      loading = true;
      notFound = false;
      found = null;
    });
    final doctor = await findClinic(controller.text);
    if (!mounted) return;
    setState(() {
      loading = false;
      found = doctor;
      notFound = doctor == null;
    });
  }

  void finish({bool solo = false}) {
    d.doctor = solo ? null : found;
    Navigator.pushAndRemoveUntil(
      context,
      MaterialPageRoute(builder: (_) => DonePage(data: d)),
      (_) => false,
    );
  }

  bool isPasswordHidden = true;

  // الحالات الديناميكية للتحقق
  bool hasMinLength = false;
  bool hasNumberAndSymbol = false;
  int strengthLevel = 0; // من 0 إلى 4

  // ألوان التصميم
  static const Color primaryGreen = Color(0xFF03662A);
  static const Color lightGreenBadge = Color(0xFFC7F9D0);
  static const Color darkText = Color(0xFF131B2E);
  static const Color subtitleText = Color(0xFF6B7280);
  static const Color inputBorderColor = Color(0xFFD1D5DB);
  static const Color inactiveGrey = Color(0xFFE5E7EB);

  // دالة فحص قوة الباسورد وتحديث الشروط ديناميكياً
  void _checkPasswordStrength(String password) {
    bool lengthValid = password.length >= 8;
    bool hasNumber = RegExp(r'[0-9]').hasMatch(password);
    bool hasSymbol = RegExp(r'[!@#$%^&*(),.?":{}|<>]').hasMatch(password);
    bool hasLetters = RegExp(r'[a-zA-Z]').hasMatch(password);

    int score = 0;
    if (password.isNotEmpty) {
      if (password.length >= 6) score++;
      if (lengthValid) score++;
      if (hasLetters && hasNumber) score++;
      if (hasNumber && hasSymbol) score++;
    }

    setState(() {
      hasMinLength = lengthValid;
      hasNumberAndSymbol = hasNumber && hasSymbol;
      strengthLevel = score;
    });
  }

  // نص قوة الباسورد
  String get strengthText {
    if (strengthLevel == 0) return "";
    switch (strengthLevel) {
      case 1:
        return "Weak";
      case 2:
        return "Fair";
      case 3:
        return "Good";
      case 4:
        return "Strong";
      default:
        return "";
    }
  }

    switch (strengthLevel) {
      case 1:
        return Colors.red;
      case 2:

        return Colors.lightGreen;

  CameraController? _cam;
  String? _error;
  bool _flash = false;
  int _mode = 0;
  XFile? _picked;

  )..repeat(reverse: true);

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _initCamera();
  }

  Future<void> _initCamera() async {
    try {
      final cams = await availableCameras();
      if (cams.isEmpty) {
        if (mounted) setState(() => _error = 'No camera found on this device');
        return;
      }
      final back = cams.firstWhere(
        (c) => c.lensDirection == CameraLensDirection.back,
        orElse: () => cams.first,
      );
      final ctrl = CameraController(
        back,
        ResolutionPreset.high,
        enableAudio: false,
      );
      await ctrl.initialize();
      if (!mounted) {
        await ctrl.dispose();
        return;
      }
      setState(() {
        _cam = ctrl;
        _error = null;
      });
    } on CameraException catch (e) {
      if (!mounted) return;
      setState(
        () => _error = e.code.contains('Denied')
            ? 'Camera permission denied.\nEnable it from system settings.'
            : (e.description ?? 'Camera error'),
      );
    }
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    final c = _cam;
    if (c == null || !c.value.isInitialized) return;
    if (state == AppLifecycleState.inactive) {
      setState(() => _cam = null);
      c.dispose();
    } else if (state == AppLifecycleState.resumed) {
      _initCamera();
    }
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    _scan.dispose();
    _cam?.dispose();
    super.dispose();
  }

  Future<void> _toggleFlash() async {
    final c = _cam;
    if (c == null || !c.value.isInitialized) return;
    _flash = !_flash;
    await c.setFlashMode(_flash ? FlashMode.torch : FlashMode.off);
    if (mounted) setState(() {});
  }

  Future<void> _capture() async {
    final picked = _picked;
    if (picked != null) {
      _onImage(picked.path);
      return;
    }
    final c = _cam;
    if (c == null || !c.value.isInitialized || c.value.isTakingPicture) return;
    HapticFeedback.mediumImpact();
    final file = await c.takePicture();
    if (!mounted) return;
    _onImage(file.path);
  }

  void _onImage(String path) {
    // TODO: send `path` to your AI analysis pipeline.
    _snack('Image ready for analysis');
  }

  void _snack(String msg) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(content: Text(msg), behavior: SnackBarBehavior.floating),
      );
  }

  Future<void> _pickFromGallery() async {
    try {
      final f = await ImagePicker().pickImage(
        source: ImageSource.gallery,
        imageQuality: 90,
      );
      if (f != null && mounted) setState(() => _picked = f);
    } catch (_) {
      if (mounted) _snack('Could not open the gallery');
    }
  }

  Future<void> _openSearch() async {
    final food = await showModalBottomSheet<FoodItem>(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (_) => const FoodSearchSheet(),
    );
    if (food != null && mounted)
      _snack('${food.name} · ${food.kcal} kcal added');
  }

  Widget _preview() {
    final picked = _picked;
    if (picked != null) {
      return Image.file(
        File(picked.path),
        fit: BoxFit.cover,
        width: double.infinity,
        height: double.infinity,

    final c = _cam;

          gradient: LinearGradient(
            begin: Alignment.topCenter,
            end: Alignment.bottomCenter,

          ),
        ),
        alignment: Alignment.center,
        padding: const EdgeInsets.all(40),

            ? const SizedBox.shrink()
            : Text(
                _error!,
                textAlign: TextAlign.center,
                style: ts(12, color: Colors.white70),
              ),
      );
    }

      child: SizedBox.expand(
        child: FittedBox(
          fit: BoxFit.cover,
          child: SizedBox(
            width: s.height,
            height: s.width,

          ),
        ),
      ),
    );

class _BracketPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final p = Paint()
      ..color = const Color(0xFF22E066)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 3
      ..strokeCap = StrokeCap.round;
    const len = 26.0;
    const r = 14.0;
    final w = size.width, h = size.height;
    final path = Path()
      ..moveTo(0, len)
      ..lineTo(0, r)
      ..quadraticBezierTo(0, 0, r, 0)
      ..lineTo(len, 0)
      ..moveTo(w - len, 0)
      ..lineTo(w - r, 0)
      ..quadraticBezierTo(w, 0, w, r)
      ..lineTo(w, len)
      ..moveTo(w, h - len)
      ..lineTo(w, h - r)
      ..quadraticBezierTo(w, h, w - r, h)
      ..lineTo(w - len, h)
      ..moveTo(len, h)
      ..lineTo(r, h)
      ..quadraticBezierTo(0, h, 0, h - r)
      ..lineTo(0, h - len);
    canvas.drawPath(path, p);
  }

  @override
  bool shouldRepaint(_) => false;
}
